### 2.2 单行合并算法（核心）

单行合并分三步：**压缩（去 0）→ 相邻等值合并 → 补 0**。

```python
def _merge_line(values: list[int]) -> tuple[list[int], int, list[tuple[int, ...]]]:
    """把一行向左合并。

    values: 长度为 4 的一行，按「移动方向前端在前」排列
    返回: (合并后的行, 本行得分, 每个输出格的来源索引组)
    """
    nz = [i for i, v in enumerate(values) if v != 0]   # 非零元素的原始索引
    out, groups, gained = [], [], 0
    k = 0
    while k < len(nz):
        i = nz[k]
        if k + 1 < len(nz) and values[i] == values[nz[k + 1]]:
            j = nz[k + 1]
            merged = values[i] * 2
            out.append(merged)
            groups.append((i, j))
            gained += merged
            k += 2                        # 跳过已被合并的那一块
        else:
            out.append(values[i])
            groups.append((i,))
            k += 1
    while len(out) < len(values):         # 补 0 到满行
        out.append(0)
        groups.append(())
    return out, gained, groups
```

**关键点**：`k += 2` 保证「每块本次只能合并一次」。若写成滑动窗口反复比较，`[2,2,4]` 会被错误合并成 `[8]`。

### 2.3 移动：四方向复用同一逻辑

```python
from dataclasses import dataclass, field

@dataclass
class MoveResult:
    changed: bool = False                          # 是否有效移动
    gained: int = 0                                # 本次得分
    slides: list = field(default_factory=list)     # [(value, (r0,c0), (r1,c1))]
    merges: list = field(default_factory=list)     # [(r, c, new_value)]
    merged_from: list = field(default_factory=list) # 被合并消失的坐标 [(r,c)]
```

```python
@staticmethod
def _lines(direction: Direction) -> list[list[tuple[int, int]]]:
    """按移动方向，把棋盘拆成 4 条线，每条线的第一个元素是「最靠近目标边界」的格。"""
    s = GRID_SIZE
    if direction == Direction.LEFT:
        return [[(r, c) for c in range(s)] for r in range(s)]
    if direction == Direction.RIGHT:
        return [[(r, c) for c in range(s - 1, -1, -1)] for r in range(s)]
    if direction == Direction.UP:
        return [[(r, c) for r in range(s)] for c in range(s)]
    return [[(r, c) for r in range(s - 1, -1, -1)] for c in range(s)]


def move(self, direction: Direction) -> MoveResult:
    result = MoveResult()
    new_grid = [[0] * GRID_SIZE for _ in range(GRID_SIZE)]

    for line in self._lines(direction):
        values = [self.grid[r][c] for r, c in line]
        if not any(values):
            continue
        out, gained, groups = _merge_line(values)
        result.gained += gained

        for k, srcs in enumerate(groups):
            if not srcs:
                continue
            tr, tc = line[k]
            new_grid[tr][tc] = out[k]
            # 位移记录（供动画使用）
            for gi in srcs:
                sr, sc = line[gi]
                result.slides.append((values[gi], (sr, sc), (tr, tc)))
            if len(srcs) > 1:              # 发生合并
                result.merges.append((tr, tc, out[k]))
                result.merged_from.append(line[srcs[1]])

    result.changed = new_grid != self.grid
    if result.changed:
        self.grid = new_grid
    return result
```

**逐向验证表**（可作为单元测试断言的依据）：

| 输入行（左→右） | 左移结果 | 得分 | 说明 |
| --- | --- | --- | --- |
| `[2,0,2,0]` | `[4,0,0,0]` | +4 | 隔空合并 |
| `[2,2,2,0]` | `[4,2,0,0]` | +4 | 前端优先配对 |
| `[2,2,2,2]` | `[4,4,0,0]` | +8 | 两组独立合并 |
| `[4,4,8,8]` | `[8,16,0,0]` | +24 | 合并结果不再二次合并 |
| `[4,0,0,4]` | `[8,0,0,0]` | +8 | 两端合并 |
| `[0,0,0,0]` | 不变 | 0 | 无效移动 |
| `[2,4,2,4]` | 不变 | 0 | 无相邻等值，无效移动 |
| `[1024,1024,0,0]` | `[2048,0,0,0]` | +2048 | 数值溢出无需处理，Python 大整数 |

### 2.4 生成、判定与重置

```python
import random

def spawn(self):
    empties = [(r, c) for r in range(GRID_SIZE) for c in range(GRID_SIZE)
               if self.grid[r][c] == 0]
    if not empties:
        return None
    r, c = random.choice(empties)
    self.grid[r][c] = 4 if random.random() < SPAWN_4_PROB else 2
    return (r, c)


def has_value(self, value: int) -> bool:
    return any(v == value for row in self.grid for v in row)


def is_game_over(self) -> bool:
    for r in range(GRID_SIZE):
        for c in range(GRID_SIZE):
            v = self.grid[r][c]
            if v == 0:
                return False
            if c + 1 < GRID_SIZE and self.grid[r][c + 1] == v:
                return False
            if r + 1 < GRID_SIZE and self.grid[r + 1][c] == v:
                return False
    return True


def max_tile(self) -> int:
    return max(v for row in self.grid for v in row)


def reset(self) -> None:
    self.grid = [[0] * GRID_SIZE for _ in range(GRID_SIZE)]
    self.spawn()
    self.spawn()
```

> 死局判定只需检查「右邻」与「下邻」，因为相邻关系是对称的，不必重复比较四个方向。

## 3. 应用层：游戏状态（`src/game.py`）

### 3.1 状态与动作枚举

```python
from enum import Enum, auto

class GameState(Enum):
    RUNNING = auto()
    WIN = auto()
    GAME_OVER = auto()

class Action(Enum):
    LEFT = auto(); RIGHT = auto(); UP = auto(); DOWN = auto()
    RESTART = auto(); UNDO = auto(); CONTINUE = auto()
```

### 3.2 Game 类

```python
class Game:
    def __init__(self, best: int = 0):
        self.board = Board()
        self.score = 0
        self.best = best
        self.state = GameState.RUNNING
        self.win_shown = False          # 通关提示只弹一次
        self._undo_stack = []           # 只保留最近 1 步
        self.last_result: MoveResult = MoveResult()   # 空结果表示尚无有效一手
        self.last_new_tile = None
        self.reset()                    # 构造即开局

    # —— 生命周期 ——
    def reset(self):
        self.board.reset()
        self.score = 0
        self.state = GameState.RUNNING
        self.win_shown = False
        self._undo_stack.clear()
        self.last_result = MoveResult()
        self.last_new_tile = None

    # —— 对外唯一入口 ——
    def handle_action(self, action: Action):
        if action is Action.RESTART:
            self.reset()
            return
        if action is Action.UNDO:
            self.undo()
            return
        if action is Action.CONTINUE:
            if self.state is GameState.WIN:
                self.state = GameState.RUNNING
            return
        if self.state is not GameState.RUNNING:
            return                      # WIN / GAME_OVER 期间忽略方向键

        direction = {Action.LEFT: Direction.LEFT, Action.RIGHT: Direction.RIGHT,
                     Action.UP: Direction.UP, Action.DOWN: Direction.DOWN}.get(action)
        if direction is not None:
            self._play(direction)

    # —— 核心一步 ——
    def _play(self, direction: Direction):
        snapshot = (self.board.clone().grid, self.score)
        result = self.board.move(direction)
        self.last_result = result       # 记录本手；无效时 changed=False，避免重播上一步动画
        if not result.changed:
            self.last_new_tile = None
            return                      # 无效移动：不生成、不加分、不撤销
        self._undo_stack.append(snapshot)
        self._undo_stack = self._undo_stack[-1:]

        self.score += result.gained
        self.best = max(self.best, self.score)
        self.last_new_tile = self.board.spawn()

        if self.board.has_value(WIN_VALUE) and not self.win_shown:
            self.win_shown = True
            self.state = GameState.WIN
        elif self.board.is_game_over():
            self.state = GameState.GAME_OVER

    def undo(self):
        # 撤销仅在 RUNNING 状态可用：WIN 浮层与 GAME_OVER 下不允许回退棋盘
        if not self._undo_stack or self.state is not GameState.RUNNING:
            return
        grid, score = self._undo_stack.pop()
        self.board.grid = [row[:] for row in grid]
        self.score = score
        self.state = GameState.RUNNING
        self.last_result = MoveResult()
        self.last_new_tile = None

    def snapshot(self) -> "GameSnapshot": ...
```

### 3.3 关于统计口径的三点约定

1. **最高分只在分数本地升高时刷新**，`reset()` 不回退 `best`。
2. **落盘由 `main.py` 统一负责**：切换设置、登记成绩、确认退出时调用 `storage.save_profile()`；模型层不写盘，便于单测。
3. **通关后再死局**：`win_shown` 保持 True，状态直接从 RUNNING 到 GAME_OVER，不再弹通关框。

### 3.4 快照（供渲染层消费的只读结构）

```python
@dataclass(frozen=True)
class GameSnapshot:
    grid: list            # 4×4 值矩阵
    score: int
    best: int
    state: GameState
    last_gain: int
```

采用 `frozen=True` + 深拷贝 grid（`[row[:] for row in grid]`），确保渲染层无法改坏模型。

```python
def snapshot(self) -> GameSnapshot:
    """导出只读快照；grid 必须深拷贝，避免渲染层意外改到模型。"""
    return GameSnapshot(
        grid=[row[:] for row in self.board.grid],
        score=self.score,
        best=self.best,
        state=self.state,
        last_gain=self.last_result.gained,
    )
```

### 4.1 存档 `src/storage.py`

存档结构（`%APPDATA%\2048Desktop\save.json`）：

```python
MAX_RECORDS = 10

@dataclass
class Record:
    name: str
    score: int
    time: str = ""


@dataclass
class Profile:
    best: int = 0
    fullscreen: bool = False
    leaderboard: list[Record] = field(default_factory=list)


def load_profile() -> Profile: ...
def save_profile(profile: Profile) -> None: ...
def qualifies(profile: Profile, score: int) -> bool: ...
def insert_record(profile: Profile, name: str, score: int) -> int | None: ...
```

约定：

- **容错**：文件缺失、JSON 乱码、字段类型不对一律降级为默认 `Profile`；兼容旧版只有 `{"best": n}` 的文件。
- **原子写**：先写 `save.tmp` 再 `os.replace`，避免写一半损坏；失败不影响游戏。
- **排行榜规则**：`score <= 0` 不入榜；满 10 条时需严格高于末位；同分按先到先得（稳定排序）；`insert_record` 返回 0 起名次，未进榜返回 `None`。

### 4.2 资源路径 `src/resource.py`

```python
import os, sys

def resource_path(*parts: str) -> str:
    if getattr(sys, "frozen", False):            # PyInstaller 打包后
        base = getattr(sys, "_MEIPASS")           # 私有属性，用 getattr 避免静态检查告警
    else:                                        # 源码运行：src/ 的上一级即工程根
        base = os.path.dirname(os.path.dirname(os.path.abspath(__file__)))
    return os.path.join(base, *parts)
```

**唯一入口原则**：所有 `assets/` 下的资源必须经此函数拼接，禁止出现 `"./assets/..."` 字面量。

```python
def cell_rect(r: int, c: int) -> tuple[int, int, int, int]:
    """第 (r, c) 格在窗口中的矩形，绘制与点击检测共用，避免两处坐标漂移。"""
    step = CELL_PX + GRID_PADDING

            BOARD_Y + GRID_PADDING + r * step,
```

### 5.2 坐标计算

棋盘左上角 `(BOARD_X, BOARD_Y) = (40, 140)`；第 (r, c) 格的矩形：

```
x = BOARD_X + GRID_PADDING + c * (CELL_PX + GRID_PADDING)
y = BOARD_Y + GRID_PADDING + r * (CELL_PX + GRID_PADDING)
w = h = CELL_PX
```

`CELL_PX = (480 - 12 * 5) // 4 = 105`，棋盘右边界 `40 + 480 = 520`，窗口左右各留 40 px 余量（40 + 480 + 40 = 560）；该公式封装为 `theme.cell_rect(r, c)`，绘制与点击检测都调它。

### 5.3 渲染器 `src/ui/renderer.py`

绘制顺序（每帧严格按此顺序，后者覆盖前者）：

1. 背景填充 `BG_COLOR`
2. 标题区文字与分数框
3. 棋盘底板（圆角矩形 `BOARD_COLOR`）
4. 16 个空格
5. 静态方块（未处于动画中的）
6. 动画方块（插值位置）与新块弹出
7. 操作区按钮与提示
8. 浮层遮罩 + 提示卡（仅 WIN / GAME_OVER 状态）

关键绘制函数：


点击检测：`Renderer.hit_test(pos, state) -> Action | None` 识别游戏内热点——

- 操作区「新游戏」按钮 → 打开二次确认弹窗（不直接重开）；
- WIN 浮层：「继续挑战」→ `Action.CONTINUE`，「重新开始」→ `Action.RESTART`；
- GAME_OVER 浮层：「再来一局」→ `Action.RESTART`；「登记成绩」（`hit_record`，仅可进榜时显示）→ 打开昵称输入。

二次确认弹窗由 `Renderer.hit_dialog(pos) -> 'confirm' | 'cancel' | None` 识别：「确认」执行对应操作，「取消」或点空白关闭弹窗。弹窗状态是 `main.py` 的 UI 局部变量（`dialog: None | 'restart' | 'exit'`），不进入模型。

命中其他区域返回 `None`。按钮矩形与绘制时共用同一份常量（见 `theme.py`），避免视觉位置与点击区域错位。

昵称输入使用 `pygame.key.start_text_input()` + `pygame.TEXTINPUT` 事件，支持输入法提交中文；长度上限 `config.MAX_NAME_LEN`。

输入体验细节：

- 输入框内绘制 500ms 闪烁光标，内容超宽时右对齐并裁剪到框内，保证光标可见；
- `pygame.TEXTEDITING` 的预编辑文本以橙色下划线实时显示（未上屏的拼音）；
- `open_entry()` 时调用 `pygame.key.set_text_input_rect()`，矩形经 `display.canvas_rect_to_screen()` 换算，候选框定位在输入框处；全屏/窗口切换后重新定位。

排行榜列为：名次 / 昵称（超长以省略号截断）/ 分数 / 日期（`MM-DD HH:MM`，缺失时显示 `—`）。

显示层 `Display`（`main.py`）负责窗口/全屏与坐标映射：

- 固定 560×720 `canvas`，所有绘制与点击热区都基于画布坐标；
- 全屏用 `set_mode((0, 0), FULLSCREEN)` 采用桌面分辨率，`smoothscale` 等比缩放居中、黑边填充；
- 鼠标事件经 `display.to_canvas(pos)` 反算，落在黑边上返回 `None`（忽略）；
- 不使用 `pygame.SCALED`（部分环境 `failed to create renderer`）。

### 5.5 动画 `src/ui/animator.py`

动画不改变逻辑，只影响绘制。数据：

```python
class Animator:
    def __init__(self, duration_ms=ANIM_MS):
        self.duration = duration_ms
        self.elapsed = duration_ms        # 初始为「已完成」，表示无动画
        self.result: MoveResult | None = None
        self.new_pos = None

    @property
    def active(self) -> bool:
        return self.elapsed < self.duration

    @property
    def progress(self) -> float:
        return min(1.0, self.elapsed / self.duration)

    def start(self, result: MoveResult, new_pos: tuple[int, int] | None):
        self.result = result
        self.new_pos = new_pos
        self.elapsed = 0

    def stop(self):
        """清空动画，重开/撤销/继续挑战/退出确认时调用，避免旧动画叠到新局面上。"""
        self.result = None
        self.new_pos = None
        self.elapsed = self.duration

    def update(self, dt_ms: int):
        if self.active:
            self.elapsed += dt_ms
```

绘制策略（`progress = p`）：

| 阶段 | p 区间 | 表现 |
| --- | --- | --- |
| 滑动 | 0 ～ 0.7 | 源方块按 `ease_out_cubic(p / 0.7)` 从起点矩形插值到终点矩形 |
| 合并 | 0.7 ～ 1.0 | 目标格绘制合并后的值，缩放系数 `1.0 + 0.15 * sin(π * q)`；合并源在前 35% 内收缩消失 |
| 新块 | 0.5 ～ 1.0 | 缩放系数从 0.6 → 1.0，与滑动后半程重叠，减少延迟感 |

动画只影响绘制：重开、撤销、继续挑战、打开退出确认等操作会先调用 `Animator.stop()` 清空未播完的动画，避免旧动画叠加到新局面上。

```python
def ease_out_cubic(t: float) -> float:
    return 1 - (1 - t) ** 3

def lerp(a, b, t): return a + (b - a) * t
```

实现要点：
- 动画期间，被合并掉的源方块**仍然绘制**（否则出现「闪空」）；目标格在合并阶段才绘制。
- 动画结束后（`active == False`）自动清空 `result`，回到纯静态绘制，避免残留。
- 多格连锁滑动（如 `[0,0,0,2]` 左移）因为 `slides` 已记录每块的起终点，插值天然正确。

## 6. 入口与主循环（`main.py`）

```python
import os, sys
import pygame

from src import config, storage
from src.game import Game, Action, GameState
from src.resource import resource_path
from src.ui.renderer import Renderer
from src.ui.animator import Animator

KEY_MAP = {
    pygame.K_LEFT: Action.LEFT,  pygame.K_a: Action.LEFT,
    pygame.K_RIGHT: Action.RIGHT, pygame.K_d: Action.RIGHT,
    pygame.K_UP: Action.UP,      pygame.K_w: Action.UP,
    pygame.K_DOWN: Action.DOWN,  pygame.K_s: Action.DOWN,
    pygame.K_r: Action.RESTART,
    pygame.K_u: Action.UNDO,
}


def main():
    pygame.init()
    pygame.display.set_caption(config.APP_NAME)
    try:
        pygame.display.set_icon(pygame.image.load(resource_path("assets", "icon.png")))
    except (pygame.error, FileNotFoundError):
        pass                                    # 图标缺失不阻断启动

    profile = storage.load_profile()
    display = Display(profile.fullscreen)       # 固定 560×720 画布；全屏等比缩放 + 坐标映射
    clock = pygame.time.Clock()
    pygame.key.set_repeat(120, 60)              # 长按 120ms 后每 60ms 重复

    game = Game(best=profile.best)
    renderer = Renderer(resource_path("assets", "fonts", "msyh.ttc"))
    animator = Animator()
    screen_name = "menu"                        # menu / game / settings / ranking / entry / entry_exit
    dialog = None

    running = True
    while running:
        dt = clock.tick(config.FPS)
        for event in pygame.event.get():
            if event.type == pygame.QUIT:       # 点 × / Alt+F4：先确认再退出
                dialog = "exit"
            elif event.type == pygame.KEYDOWN:
                if event.key == pygame.K_ESCAPE:
                    dialog = None if dialog is not None else "exit"
                    continue
                if dialog is not None:          # 弹窗模态：方向键/撤销忽略
                    continue
                action = KEY_MAP.get(event.key)
                if action is not None:
                    _dispatch(game, animator, action)
            elif event.type == pygame.MOUSEBUTTONDOWN and event.button == 1:
                pos = display.to_canvas(event.pos)   # 全屏时反算回画布坐标，黑边点击忽略
                if pos is None:
                    continue
                if dialog is not None:
                    # 「确认」→ 重开或退出；「取消」/空白 → 关闭弹窗（见 §5.3）
                    continue
                action = Renderer.hit_test(pos, game.state)
                if action is not None:
                    _dispatch(game, animator, action)

        animator.update(dt)
        renderer.draw(display.canvas, game.snapshot(), animator)   # 其它界面见 §5.4
        if dialog is not None:
            renderer.draw_dialog(display.canvas, dialog)
        display.present()                       # 窗口 1:1；全屏等比缩放居中

    profile.best = max(profile.best, game.best)
    storage.save_profile(profile)               # 退出前落盘
    pygame.quit()
    sys.exit(0)


def _dispatch(game: Game, animator: Animator, action: Action):
    previous = game.last_result                # 记住上一手，用于判断本手是否真的执行
    if action not in _DIRECTION_ACTIONS:        # 重开/撤销/继续挑战：先终止旧动画
        animator.stop()
    game.handle_action(action)
    result = game.last_result
    if action in _DIRECTION_ACTIONS and result is not previous and result.changed:
        animator.start(result, game.last_new_tile)


if __name__ == "__main__":
    main()
```

注意：

- `_dispatch` 中只有「本手确实发生移动」才启动动画；`UNDO` / `RESTART` 不播动画。
- 动画播放期间方向键进入单步缓冲（只留最近一手），动画结束再执行；重开/撤销/继续挑战立即生效并终止动画。
- WIN / GAME_OVER 浮层在本手动画播完后才绘制，避免最后一手被遮住。
- 退出确认：点 × / Alt+F4 / ESC 都打开 `dialog="exit"`，确认后才 `save_profile()` 并退出；「取消」、点空白或再按 ESC 关闭弹窗。
- 弹窗期间是模态的：方向键、撤销、重开快捷键均忽略；`R` 仅在「重新开始」弹窗内等同确认。
- `pygame.key.set_repeat` 与 `KEYDOWN` 事件天然配合，无需自建长按计时器。
- 退出前必须 `save_profile`，保证「中途关闭仍保留纪录」；本局分数可进前 10 且未登记时，先进入昵称输入再退出。
- 界面状态机（`menu/game/settings/ranking/entry/entry_exit`）与弹窗只存在于 `main.py`，不进入模型；`entry_exit` 表示登记后直接退出；F11 随时切换全屏并持久化。
- 发布构建需在 `pygame.init()` 之前声明 DPI 感知（`ctypes.windll.shcore.SetProcessDpiAwareness(1)`），否则高分屏下文字发虚，实现见 06 §6.5。
